Инструкция подключения Google Таблиц

1. Создайте новую Google Таблицу.

2. В верхнем меню выберите Расширения → Apps Script.

3. Удалите весь код и вставьте следующий скрипт:

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  // Если таблица пустая, добавляем заголовки
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Дата", "Класс", "ФИО Ученика", "№ Урока", "Тема", "Сложность", "Оценка", "Вероятность ИИ (%)", "Что повторить"]);
  }
  
  sheet.appendRow([
    data.date,
    data.studentClass,
    data.studentName,
    data.lessonNum,
    data.topic,
    data.difficulty,
    data.score,
    data.aiUsagePercent + "%",
    data.repeatTopic
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}

4. Нажмите Начать развертывание → Новое развертывание. Тип: Веб-приложение. Доступ: Все (Anyone).

5. Скопируйте полученный URL и вставьте в поле настроек в кабинете учителя.
